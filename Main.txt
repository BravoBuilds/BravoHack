local function LoadPersistentSettings()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then return end
    local path = "BravoHack/settings.json"
    if not isfile(path) then return end
    local ok, raw = pcall(readfile, path)
    if not ok or type(raw) ~= "string" or raw == "" then return end
    local decodeOk, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(raw)
    end)
    if not decodeOk or type(data) ~= "table" then return end
    if type(data.AutoRememberKey) == "boolean" then State.Features.AutoRememberKey = data.AutoRememberKey end
    if type(data.AutoLoadConfig) == "boolean" then State.Features.AutoLoadConfig = data.AutoLoadConfig end
    if type(data.ConfigSelectedName) == "string" and data.ConfigSelectedName ~= "" then
        State.ConfigSelectedName = data.ConfigSelectedName
    end
end

local function LoadRememberedKey()
    if not State.Features.AutoRememberKey then return nil end
    if type(isfile) ~= "function" or type(readfile) ~= "function" then return nil end
    local path = "BravoHack/key.json"
    if not isfile(path) then return nil end
    local ok, raw = pcall(readfile, path)
    if not ok or type(raw) ~= "string" or raw == "" then return nil end
    local decodeOk, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(raw)
    end)
    if decodeOk and type(data) == "table" and type(data.Key) == "string" then
        return data.Key:gsub("%s+", "")
    end
    return nil
end

local function ClearRememberedKeyFile()
    for _, fnName in ipairs({"delfile", "removefile"}) do
        local fn = _G[fnName]
        if type(fn) == "function" then
            pcall(function() fn("BravoHack/key.json") end)
            return
        end
    end
end

local function GetRequestFunction()
    if type(request) == "function" then
        return request
    elseif type(http_request) == "function" then
        return http_request
    elseif type(syn) == "table" and type(syn.request) == "function" then

    end
    return nil
end

local function AuthorizeKeyWithAPI(key)
    local requestFn = GetRequestFunction()
    if not requestFn then
        return false, "HTTP requests are unavailable in this executor."
    end

    local HttpService = game:GetService("HttpService")
    local okEncode, body = pcall(function()
        return HttpService:JSONEncode({
            key = key,
            userId = tostring(LocalPlayer.UserId),
        })
    end)
    if not okEncode then
        return false, "Failed to encode API request."
    end

    local okRequest, response = pcall(function()
        return requestFn({
            Url = KeyGate.ApiURL,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
                ["Accept"] = "application/json",
            },
            Body = body,
        })
    end)

    if not okRequest or type(response) ~= "table" then
        return false, "Key API connection failed."
    end

    local statusCode = tonumber(response.StatusCode or response.Status or 0) or 0
    local responseBody = response.Body
    if type(responseBody) ~= "string" or responseBody == "" then
        return false, statusCode >= 500 and "Key API is temporarily unavailable." or "Key API returned an empty response."
    end

    local decodeOk, data = pcall(function()
        return HttpService:JSONDecode(responseBody)
    end)
    if not decodeOk or type(data) ~= "table" then
        return false, "Key API returned invalid JSON."
    end

    if statusCode == 200 and data.ok == true then
        if data.claimed == true then
            return true, "Access granted. Key is now bound to your Roblox UserId."
        end
        return true, "Access granted. Key is already bound to your Roblox UserId."
    end

    if statusCode == 404 and data.error == "INVALID_KEY" then
        return false, "Invalid key."
    end

    if statusCode == 409 and data.error == "KEY_ALREADY_USED" then
        return false, "This key is already bound to another Roblox UserId."
    end

    if statusCode >= 500 then
        return false, "Key API is temporarily unavailable."
    end

    return false, tostring(data.message or data.error or "Key authorization failed.")
end

SubmitKey = function()
    if KeyGate.InteractionBusy or State.KeyAuthorized or _G.__BravoHackUnloaded then return end

    local key = tostring(KeyGate.Input.Text or "")
    key = key:gsub("%s+", "")

    if key == "" then
        KeyGate.Status.Text = "Enter a key first."
        KeyGate.Status.TextColor3 = CONFIG.Colors.Red
        pcall(function() KeyGate.Input:CaptureFocus() end)
        return
    end

    KeyGate.InteractionBusy = true
    KeyGate.Status.Text = "Checking key with external API..."
    KeyGate.Status.TextColor3 = CONFIG.Colors.SubText

    local valid, message = AuthorizeKeyWithAPI(key)
    KeyGate.InteractionBusy = false

    if not valid then
        KeyGate.Status.Text = message or "Invalid, used, or unavailable key."
        KeyGate.Status.TextColor3 = CONFIG.Colors.Red
        return
    end

    KeyGate.Status.Text = "Access granted. Key bound to UserId " .. tostring(LocalPlayer.UserId) .. "."
    KeyGate.Status.TextColor3 = CONFIG.Colors.Green
    KeyGate.Button.Text = "Unlocked"
    if State.Features.AutoRememberKey and ConfigStore and ConfigStore.SaveRememberedKey then
        pcall(function() ConfigStore.SaveRememberedKey(key) end)
    else
        pcall(ClearRememberedKeyFile)
    end
    OpenAuthorizedMenu()

    task.defer(function()
        pcall(function() KeyGate.Gui:Destroy() end)
    end)
end

local function CopyKeyLink()
    local copied = false
    local function tryClipboard(fn)
        if type(fn) ~= "function" then return end
        local ok = pcall(function() fn(KeyGate.ApiURL) end)
        if ok then copied = true end
    end
    tryClipboard(setclipboard)
    if not copied then tryClipboard(toclipboard) end
    if not copied then tryClipboard(set_clipboard) end
    if not copied then tryClipboard(writeclipboard) end

    if copied then
        KeyGate.Status.Text = "Key API link copied. Keys are API-bound after first use."
        KeyGate.Status.TextColor3 = CONFIG.Colors.Green
    else
        KeyGate.Status.Text = "Clipboard is unavailable in this environment."
        KeyGate.Status.TextColor3 = CONFIG.Colors.SubText
    end
end

    for tabName, tabButton in pairs(SidebarButtons) do
        local selected = (tabName == State.CurrentTab)
        Tween(tabButton, TI_FAST, {
            TextColor3 = selected and CONFIG.Colors.Text or CONFIG.Colors.SubText,
            BackgroundColor3 = Color3.fromRGB(17, 18, 22),
            BackgroundTransparency = 1,
        })
    end
    local currentButton = SidebarButtons[State.CurrentTab]
    if currentButton then
        Tween(UI.Indicator, TI_MED, {

                currentButton.AbsolutePosition.X - UI.MainFrame.AbsolutePosition.X,

local function SwitchTab(name)
    if not Pages[name] or not SidebarButtons[name] then return end
    local previousPage = Pages[State.CurrentTab]
    local nextPage = Pages[name]
    if previousPage == nextPage then
        RefreshTabVisuals()
        return
    end

    previousPage.Visible = false
    State.CurrentTab = name
    UI.TabTitle.Text = name
    local basePos = PageBasePositions[name] or UDim2.new(0, 30, 0, 58)
    nextPage.Position = UDim2.new(basePos.X.Scale, basePos.X.Offset + 14, basePos.Y.Scale, basePos.Y.Offset)
    nextPage.Visible = true
    Tween(nextPage, TI_MED, { Position = basePos })
    RefreshTabVisuals()
end

    local currentValue = default

    local function updateFromAlpha(alpha)
        alpha = math.clamp(alpha, 0, 1)
        local raw = min + (max - min) * alpha
        if decimals == 0 then
            raw = math.floor(raw + 0.5)
        else
            local mult = 10 ^ decimals
            raw = math.floor(raw * mult + 0.5) / mult
        end
        currentValue = raw
        fill.Size = UDim2.new(alpha, 0, 1, 0)
        knob.Position = UDim2.new(alpha, 0, 0.5, 0)
        valueLabel.Text = tostring(currentValue)
        if callback then
            local ok, err = pcall(callback, currentValue)
            if not ok then warn("[BravoHack] Slider error ("..text.."):", err) end
        end
    end

    local function inputToAlpha(input)
        local relX = input.Position.X - track.AbsolutePosition.X
        return relX / track.AbsoluteSize.X
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateFromAlpha(inputToAlpha(input))
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateFromAlpha(inputToAlpha(input))
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    local ref = {
        Get = function() return currentValue end,
        Set = function(v) updateFromAlpha((v - min) / (max - min)) end,
    }
    table.insert(SliderRefs, { ref = ref, default = default })
    return ref
end

local function TriggerKeybind(keyCode)
    if not keyCode then return false end
    for _, entry in pairs(FeatureKeybindRefs) do
        local bound = entry.Get and entry.Get()
        if bound and bound == keyCode and entry.Toggle then
            entry.Toggle()
            return true
        end
    end
    for _, entry in pairs(ActionKeybindRefs) do
        local bound = entry.Get and entry.Get()
        if bound and bound == keyCode and entry.Trigger then
            entry.Trigger()
            return true
        end
    end
    return false
end

DestroySkeletonLines = function(objs)
    for _, line in ipairs((objs and objs.SkeletonSegments) or {}) do
        if line.Glow and line.Glow.Parent then line.Glow:Destroy() end
        if line.Frame and line.Frame.Parent then line.Frame:Destroy() end
    end
    for _, joint in ipairs((objs and objs.SkeletonJoints) or {}) do
        if joint.Frame and joint.Frame.Parent then joint.Frame:Destroy() end
    end

    if objs and objs.SkeletonHead and objs.SkeletonHead.Parent then
        objs.SkeletonHead:Destroy()
    end
    if objs then objs.SkeletonHead = nil end
end

local function GetCharacterScreenBounds(char, camera)
    if not char or not camera then return nil end
    local ok, cf, size = pcall(function() return char:GetBoundingBox() end)
    if not ok or not cf or not size then return nil end
    local half = size * 0.5
    local corners = {
        Vector3.new(-half.X, -half.Y, -half.Z), Vector3.new(-half.X, -half.Y, half.Z),
        Vector3.new(-half.X, half.Y, -half.Z), Vector3.new(-half.X, half.Y, half.Z),
        Vector3.new(half.X, -half.Y, -half.Z), Vector3.new(half.X, -half.Y, half.Z),
        Vector3.new(half.X, half.Y, -half.Z), Vector3.new(half.X, half.Y, half.Z),
    }
    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local any = false
    for _, offset in ipairs(corners) do
        local screen, onScreen = camera:WorldToViewportPoint(cf:PointToWorldSpace(offset))
        if screen.Z > 0 then
            any = true
            minX = math.min(minX, screen.X)
            minY = math.min(minY, screen.Y)
            maxX = math.max(maxX, screen.X)
            maxY = math.max(maxY, screen.Y)
        end
    end
    if not any then return nil end
    local vp = camera.ViewportSize
    minX = math.clamp(minX, 0, vp.X)
    minY = math.clamp(minY, 0, vp.Y)
    maxX = math.clamp(maxX, 0, vp.X)
    maxY = math.clamp(maxY, 0, vp.Y)
    if maxX - minX < 4 or maxY - minY < 8 then return nil end
    return minX, minY, maxX, maxY
end

local FreecamConn
local SavedCameraType, SavedCameraSubject
local SavedMouseBehavior, SavedMouseIconEnabled
local FreecamPosition = nil
local FreecamYaw = 0
local FreecamPitch = 0
local FreecamSpeed = 55
local FreecamSavedRootCFrame = nil
local FreecamSavedRootAnchored = nil
local FreecamSavedAutoRotate = nil
local FreecamSavedWalkSpeed = nil
local FreecamSavedJumpPower = nil
local FreecamChatFocused = false

local function FreezeCharacterForFreecam(forceCapture)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not hrp then return false end

    if forceCapture or FreecamSavedRootCFrame == nil then FreecamSavedRootCFrame = hrp.CFrame end
    if FreecamSavedRootAnchored == nil then FreecamSavedRootAnchored = hrp.Anchored end
    if hum then
        if FreecamSavedAutoRotate == nil then FreecamSavedAutoRotate = hum.AutoRotate end
        if FreecamSavedWalkSpeed == nil then FreecamSavedWalkSpeed = hum.WalkSpeed end
        if FreecamSavedJumpPower == nil then FreecamSavedJumpPower = hum.JumpPower end
        hum.AutoRotate = false
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum:ChangeState(Enum.HumanoidStateType.Physics)
    end
    hrp.Anchored = true
    hrp.CFrame = FreecamSavedRootCFrame
    return true
end

local function RestoreCharacterAfterFreecam()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
        if FreecamSavedRootCFrame then hrp.CFrame = FreecamSavedRootCFrame end
        if FreecamSavedRootAnchored ~= nil then hrp.Anchored = FreecamSavedRootAnchored end
    end
    if hum then
        if FreecamSavedAutoRotate ~= nil then hum.AutoRotate = FreecamSavedAutoRotate end
        if FreecamSavedWalkSpeed ~= nil then hum.WalkSpeed = FreecamSavedWalkSpeed end
        if FreecamSavedJumpPower ~= nil then hum.JumpPower = FreecamSavedJumpPower end
        if hum.Health > 0 then hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end
    FreecamSavedRootCFrame = nil
    FreecamSavedRootAnchored = nil
    FreecamSavedAutoRotate = nil
    FreecamSavedWalkSpeed = nil
    FreecamSavedJumpPower = nil
end

    local rotation = CFrame.fromOrientation(FreecamPitch, FreecamYaw, 0)
    Camera.CFrame = CFrame.new(FreecamPosition) * rotation
    Camera.Focus = CFrame.new(FreecamPosition + rotation.LookVector * 100)
end

local function FocusRobloxChat()
    ChatInputBarConfiguration = TextChatService:FindFirstChild("ChatInputBarConfiguration") or ChatInputBarConfiguration
    if ChatInputBarConfiguration and ChatInputBarConfiguration.Enabled then
        local textBox = ChatInputBarConfiguration.TextBox
        if textBox and textBox.TextEditable then
            FreecamChatFocused = true

            pcall(function() textBox:CaptureFocus() end)
            return true
        end
    end

    for _, obj in ipairs(PlayerGui:GetDescendants()) do

function EnableFreecam()
    Camera = Workspace.CurrentCamera or Camera
    if not Camera then return end
    SavedCameraType = Camera.CameraType
    SavedCameraSubject = Camera.CameraSubject
    SavedMouseBehavior = UserInputService.MouseBehavior
    SavedMouseIconEnabled = UserInputService.MouseIconEnabled

    local look = Camera.CFrame.LookVector
    FreecamPosition = Camera.CFrame.Position
    FreecamYaw = math.atan2(-look.X, -look.Z)
    FreecamPitch = math.clamp(math.asin(math.clamp(look.Y, -0.999, 0.999)), math.rad(-85), math.rad(85))

    FreezeCharacterForFreecam()
    Camera.CameraType = Enum.CameraType.Scriptable
    Camera.FieldOfView = State.Sliders.CameraFOV

    FreecamChatFocused = UserInputService:GetFocusedTextBox() ~= nil
    SetFreecamInputMode(FreecamChatFocused)
    SetFreecamCamera()

    if FreecamConn then FreecamConn:Disconnect() end
    RunService:BindToRenderStep("DeveloperFreecam", Enum.RenderPriority.Camera.Value + 5, function(dt)
        if not State.Features.Freecam then return end
        Camera = Workspace.CurrentCamera or Camera
        if not Camera then return end
        FreezeCharacterForFreecam(false)

        local focusedBox = UserInputService:GetFocusedTextBox()
        local chatFocused = FreecamChatFocused or focusedBox ~= nil
        if ChatInputBarConfiguration and ChatInputBarConfiguration.IsFocused then
            chatFocused = true
        end
        if chatFocused then
            FreecamChatFocused = true
            SetFreecamInputMode(true)
            Camera.CameraType = Enum.CameraType.Scriptable
            return
        end
        SetFreecamInputMode(false)

        local delta = UserInputService:GetMouseDelta()
        FreecamYaw -= delta.X * 0.0028
        FreecamPitch = math.clamp(FreecamPitch - delta.Y * 0.0028, math.rad(-85), math.rad(85))

        local rotation = CFrame.fromOrientation(FreecamPitch, FreecamYaw, 0)
        local move = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += rotation.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= rotation.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= rotation.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += rotation.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.yAxis end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.yAxis end
        if move.Magnitude > 0 then
            local speed = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) and FreecamSpeed * 3 or FreecamSpeed
            FreecamPosition += move.Unit * speed * dt
        end
        Camera.CameraType = Enum.CameraType.Scriptable
        Camera.CFrame = CFrame.new(FreecamPosition) * rotation
        Camera.Focus = CFrame.new(FreecamPosition + rotation.LookVector * 100)
    end)
end

function DisableFreecam()

    pcall(function() RunService:UnbindFromRenderStep("DeveloperFreecam") end)
    if FreecamConn then
        FreecamConn:Disconnect()
        FreecamConn = nil
    end

    pcall(function()
        local focused = UserInputService:GetFocusedTextBox()
        if focused then focused:ReleaseFocus() end
    end)

    if Camera then
        Camera.CameraType = SavedCameraType or Enum.CameraType.Custom
        if SavedCameraSubject and SavedCameraSubject.Parent then
            Camera.CameraSubject = SavedCameraSubject
        else
            local hum = GetHumanoid()
            if hum then Camera.CameraSubject = hum end
        end
    end

    FreecamChatFocused = false

    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    UserInputService.MouseIconEnabled = true

    RestoreCharacterAfterFreecam()
    FreecamPosition = nil

    task.defer(function()
        if not State.Features.Freecam then
            UserInputService.MouseBehavior = Enum.MouseBehavior.Default
            UserInputService.MouseIconEnabled = true
        end
    end)
end

function GetAimTargetPart(char, targetName)
    if not char then return nil end
    targetName = targetName or State.AimTargetPart or "Head"
    local part = char:FindFirstChild(targetName)
    if part and part:IsA("BasePart") then
        return part
    end
    local fallbacks = {"Head", "UpperTorso", "Torso", "LowerTorso", "HumanoidRootPart"}
    for _, name in ipairs(fallbacks) do
        local fallback = char:FindFirstChild(name)
        if fallback and fallback:IsA("BasePart") then
            return fallback
        end
    end
    return nil
end

function AimTargetVisible(targetPart, useWallCheck)
    if not targetPart or not Camera then return false end
    if useWallCheck == nil then useWallCheck = State.Features.WallCheck end
    if not useWallCheck then return true end
    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    if direction.Magnitude <= 0.01 then return true end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    local result = Workspace:Raycast(origin, direction, params)
    if not result then return true end
    local targetCharacter = targetPart:FindFirstAncestorOfClass("Model")
    return targetCharacter and result.Instance:IsDescendantOf(targetCharacter) or result.Instance == targetPart
end

    local function getRadarColor(distance)
        local range = math.max(1, State.Sliders.RadarRange)
        local closeness = 1 - math.clamp(distance / range, 0, 1)
        return Color3.fromHSV(0.33 * (1 - closeness), 0.95, 1)
    end

    RadarUpdate = function()
        radarFrame.Visible = State.Features.Radar

        if not State.Features.Radar or not Camera then
            for _, marker in pairs(markers) do
                marker.Visible = false
            end
            return
        end

        local cameraCF = Camera.CFrame
        local camPos = cameraCF.Position
        local forward = Vector3.new(cameraCF.LookVector.X, 0, cameraCF.LookVector.Z)
        local right = Vector3.new(cameraCF.RightVector.X, 0, cameraCF.RightVector.Z)

        if forward.Magnitude < 0.001 or right.Magnitude < 0.001 then
            return
        end

        forward = forward.Unit
        right = right.Unit

        local active = {}
        local maxOffset = 52

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local char = plr.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")

                if root and hum and hum.Health > 0 then
                    local flat = Vector3.new(root.Position.X - camPos.X, 0, root.Position.Z - camPos.Z)
                    local distance = flat.Magnitude

                    if distance <= math.max(1, State.Sliders.RadarRange) then
                        local rightAmount = right:Dot(flat)
                        local forwardAmount = forward:Dot(flat)
                        local x = rightAmount
                        local y = -forwardAmount
                        local magnitude = math.sqrt(x * x + y * y)

                        if magnitude > maxOffset then
                            local scale = maxOffset / magnitude
                            x *= scale
                            y *= scale
                        end

                        local marker = getMarker(plr)
                        marker.Visible = true
                        marker.Position = UDim2.fromOffset(Camera.ViewportSize.X * 0.5 + x, Camera.ViewportSize.Y * 0.5 + y)
                        marker.TextColor3 = getRadarColor(distance)
                        marker.Rotation = math.deg(math.atan2(x, -y))
                        active[plr] = true
                    end
                end
            end
        end

        for plr, marker in pairs(markers) do
            if not active[plr] then
                marker.Visible = false
            end
        end
    end

    RadarFrame = radarFrame
    State.UpdateRadar = RadarUpdate

        if marker then
            marker:Destroy()
        end
        markers[plr] = nil
    end)

local MiscUI = {}

local function GetPlayerTools(plr)
    local names, seen = {}, {}
    if not plr then return names end

    local backpack = plr:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, obj in ipairs(backpack:GetChildren()) do
            if obj:IsA("Tool") and not seen[obj.Name] then
                seen[obj.Name] = true
                table.insert(names, obj.Name)
            end
        end
    end

    local char = plr.Character
    if char then
        for _, obj in ipairs(char:GetChildren()) do
            if obj:IsA("Tool") and not seen[obj.Name] then
                seen[obj.Name] = true
                table.insert(names, obj.Name .. " [equipped]")
            end
        end
    end

    table.sort(names)
    return names
end

local function FindCrosshairPlayer()
    if not Camera then return nil end
    local vp = Camera.ViewportSize
    local ray = Camera:ViewportPointToRay(vp.X * 0.5, vp.Y * 0.5)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character}

    local result = Workspace:Raycast(ray.Origin, ray.Direction * 2000, params)
    if not result or not result.Instance then return nil end
    local model = result.Instance:FindFirstAncestorOfClass("Model")
    return model and Players:GetPlayerFromCharacter(model) or nil
end

State.TeleportToSelectedPlayer = function()
    local target = State.SelectedPlayer
    if not target or target == LocalPlayer or not target.Parent then
        if Notify then Notify("Teleport", "Select a valid player first.", true) end
        return false
    end

    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myChar or not myRoot then
        if Notify then Notify("Teleport", "Your character is not loaded yet.", true) end
        return false
    end

    local targetChar = target.Character
    local targetRoot = targetChar and (targetChar:FindFirstChild("HumanoidRootPart") or targetChar:WaitForChild("HumanoidRootPart", 2))
    if not targetRoot then
        if Notify then Notify("Teleport", "The selected player has no HumanoidRootPart.", true) end
        return false
    end

    local destination = targetRoot.CFrame * CFrame.new(3, 0, 0)
    myChar:PivotTo(destination)
    myRoot.AssemblyLinearVelocity = Vector3.zero
    myRoot.AssemblyAngularVelocity = Vector3.zero

    
    task.spawn(function()
        for i = 1, 6 do
            task.wait(0.05)
            if not target.Parent then break end
            local currentRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
            local ownRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if not currentRoot or not ownRoot then break end
            LocalPlayer.Character:PivotTo(currentRoot.CFrame * CFrame.new(3, 0, 0))
            ownRoot.AssemblyLinearVelocity = Vector3.zero
            ownRoot.AssemblyAngularVelocity = Vector3.zero
        end
    end)

    if Notify then Notify("Teleport", "Teleported to @" .. target.Name .. ".", true) end
    return true
end

local function ApplySpectateTarget(plr)
    if not plr or plr == LocalPlayer then return false end
    Camera = Workspace.CurrentCamera or Camera
    if not Camera then return false end

    local hum = char and char:FindFirstChildOfClass("Humanoid")

        return false
    end

    Camera.CameraType = Enum.CameraType.Custom
    Camera.CameraSubject = hum

end

State.StopSpectatingSelectedPlayer = function()
    State.Features.Spectate = false

    if SpectateCharacterConn then
        SpectateCharacterConn:Disconnect()
        SpectateCharacterConn = nil
    end

    Camera = Workspace.CurrentCamera or Camera
    if Camera then
        Camera.CameraType = SpectateSavedCameraType or Enum.CameraType.Custom
        if SpectateSavedCameraSubject and SpectateSavedCameraSubject.Parent then
            Camera.CameraSubject = SpectateSavedCameraSubject
        else
            local hum = GetHumanoid()
            if hum then Camera.CameraSubject = hum end
        end
    end

    SpectateTarget = nil
    SpectateSavedCameraType = nil
    SpectateSavedCameraSubject = nil
end

State.StartSpectatingSelectedPlayer = function()
    local target = State.SelectedPlayer
    if not target or target == LocalPlayer or not target.Parent then
        State.Features.Spectate = false
        if Notify then Notify("Spectate", "Select a player in the Player List first.", true) end
        return false
    end

    if State.Features.Freecam then
        State.Features.Spectate = false
        if Notify then Notify("Spectate", "Disable Freecam first.", true) end
        return false
    end

    Camera = Workspace.CurrentCamera or Camera
    if not Camera then
        State.Features.Spectate = false
        return false
    end

    if not State.Features.Spectate or SpectateTarget ~= target then
        SpectateSavedCameraType = Camera.CameraType
        SpectateSavedCameraSubject = Camera.CameraSubject
    end

    State.Features.Spectate = true
    SpectateTarget = target

    if SpectateCharacterConn then
        SpectateCharacterConn:Disconnect()
        SpectateCharacterConn = nil
    end

    SpectateCharacterConn = target.CharacterAdded:Connect(function(char)
        if not State.Features.Spectate or SpectateTarget ~= target then return end
        local hum = char:WaitForChild("Humanoid", 5)
        if hum and Camera then
            Camera.CameraType = Enum.CameraType.Custom
            Camera.CameraSubject = hum
        end
    end)

    if not ApplySpectateTarget(target) then
        
        local char = target.Character or target.CharacterAdded:Wait()
        local hum = char and char:FindFirstChildOfClass("Humanoid") or (char and char:WaitForChild("Humanoid", 2))
        if hum and Camera then
            Camera.CameraType = Enum.CameraType.Custom
            Camera.CameraSubject = hum
        else
            State.StopSpectatingSelectedPlayer()
            if Notify then Notify("Spectate", "This player currently has no Humanoid.", true) end
            return false
        end
    end

    if Notify then Notify("Spectate", "Now spectating " .. target.Name .. ".", true) end
    return true
end

State.UpdateInventoryView = function()
    if not MiscUI.InventoryLabel then return end
    if not State.Features.InventoryView then
        return
    end

    local target = State.SelectedPlayer
    if not target or target == LocalPlayer or not target.Parent then
        MiscUI.InventoryLabel.Text = "Selected: none"
        return
    end

    local inventory = target:FindFirstChild("Inventory") or target:FindFirstChild("Inevtorie")
    if not inventory then
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\nInventory: not found"
        return
    end

    local items = {}
    for _, item in ipairs(inventory:GetChildren()) do
        local valueObject = item:FindFirstChild("Value")
        local value = nil

        if item:IsA("IntValue") or item:IsA("NumberValue") then
            value = item.Value
        elseif valueObject and (valueObject:IsA("IntValue") or valueObject:IsA("NumberValue")) then
            value = valueObject.Value
        end

        if value and value >= 1 then
            if math.floor(value) == value then
                table.insert(items, string.format("%s  x%d", item.Name, value))
            else
                table.insert(items, string.format("%s  x%.2f", item.Name, value))
            end
        end
    end

    table.sort(items, function(a, b) return a:lower() < b:lower() end)

    if #items == 0 then
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\nNo inventory items with Value >= 1"
    else
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\n" .. table.concat(items, "\n")
    end
end

        end

            if Notify then Notify("Lua", "loadstring is unavailable in this executor.", true) end
            return false
        end

        local compileOk, chunk = pcall(loadstring, source)
        if not compileOk or type(chunk) ~= "function" then
            if Notify then Notify("Lua", "Compile error: " .. tostring(chunk), true) end
            return false
        end

        local runOk, err = pcall(chunk)
        if Notify then
            Notify("Lua", runOk and ("Ran " .. selected.name .. ".") or ("Run error in " .. selected.name .. ": " .. tostring(err)), true)
        end
        return runOk
    end

    UI.LuaRunner.CloseAll = function()
        for name, gui in pairs(UI.LuaRunner.Guis) do
            pcall(function()
                if gui and gui.Parent then gui:Destroy() end
            end)
            UI.LuaRunner.Guis[name] = nil
        end
        pcall(function()
            local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui")
            if pg then
                local inventoryGui = pg:FindFirstChild("BravoHackInventoryLua")
                if inventoryGui then inventoryGui:Destroy() end
            end
        end)
    end
end

local MiscWindowSizes = {}
local MiscWindowBusy = {}

SetMiscWindowVisible = function(frame, visible)
    if not frame or not frame.Parent then return end

    local originalSize = MiscWindowSizes[frame]
    if not originalSize then
        originalSize = frame.Size
        MiscWindowSizes[frame] = originalSize
    end

    frame:SetAttribute("ShouldBeVisible", visible)
    if MiscWindowBusy[frame] then return end

    if visible then
        MiscWindowBusy[frame] = true
        frame.Visible = true
        frame.Size = UDim2.fromOffset(0, 0)
        local tw = Tween(frame, TI_SLOW, { Size = originalSize })
        tw.Completed:Connect(function()
            MiscWindowBusy[frame] = nil
            if frame.Parent and frame:GetAttribute("ShouldBeVisible") then
                frame.Size = originalSize
            end
        end)
    else
        MiscWindowBusy[frame] = true
        local tw = Tween(frame, TI_MED, { Size = UDim2.fromOffset(0, 0) })
        tw.Completed:Connect(function()
            MiscWindowBusy[frame] = nil
            if frame.Parent and not frame:GetAttribute("ShouldBeVisible") then
                frame.Visible = false
                frame.Size = originalSize
            end
        end)
    end
end

SavedConfigs, ConfigStore = {}, {
    Root = "BravoHack",
    Configs = "BravoHack/Configs",
    Version = "v1.0.8",
    KeyFile = "BravoHack/key.json",
    MetaFile = "BravoHack/settings.json",
}

ConfigStore.SafeName = function(name)
    return tostring(name or "Default"):gsub("[^%w%-%_ ]", "_")
end

ConfigStore.Path = function(name)
    return ConfigStore.Configs .. "/" .. ConfigStore.SafeName(name) .. ".json"
end

ConfigStore.Ensure = function()
    if type(makefolder) ~= "function" or type(writefile) ~= "function" or type(readfile) ~= "function" then
        return false
    end

    pcall(function()
        if type(isfolder) ~= "function" or not isfolder(ConfigStore.Root) then
            makefolder(ConfigStore.Root)
        end
    end)

    pcall(function()
        if type(isfolder) ~= "function" or not isfolder(ConfigStore.Configs) then
            makefolder(ConfigStore.Configs)
        end
    end)

    pcall(function()
        writefile(ConfigStore.Root .. "/version.txt", ConfigStore.Version)
    end)

    return true
end

ConfigStore.SaveRememberedKey = function(key)
    if not ConfigStore.Ensure() then return false end
    if type(writefile) ~= "function" then return false end
    local ok, encoded = pcall(function()
        return game:GetService("HttpService"):JSONEncode({Key = tostring(key)})
    end)
    if not ok then return false end
    return pcall(function() writefile(ConfigStore.KeyFile, encoded) end)
end

ConfigStore.ClearRememberedKey = function()
    if type(delfile) == "function" then pcall(delfile, ConfigStore.KeyFile) end
    if type(removefile) == "function" then pcall(removefile, ConfigStore.KeyFile) end
end

ConfigStore.SaveMeta = function()
    if not ConfigStore.Ensure() or type(writefile) ~= "function" then return false end
    local data = {
        AutoRememberKey = State.Features.AutoRememberKey == true,
        AutoLoadConfig = State.Features.AutoLoadConfig == true,
        ConfigSelectedName = State.ConfigSelectedName or "Default",
    }
    local ok, encoded = pcall(function()
        return game:GetService("HttpService"):JSONEncode(data)
    end)
    if not ok then return false end
    return pcall(function() writefile(ConfigStore.MetaFile, encoded) end)
end

ConfigStore.List = function()
    local names = {}
    if type(listfiles) == "function" then
        local ok, files = pcall(function()
            return listfiles(ConfigStore.Configs)
        end)
        if ok and type(files) == "table" then
            for _, path in ipairs(files) do
                local name = tostring(path):match("([^/\\]+)%.json$")
                if name and name ~= "" then
                    table.insert(names, name)
                end
            end
        end
    end
    table.sort(names, function(a, b) return a:lower() < b:lower() end)
    if #names == 0 then names[1] = "Default" end
    return names
end

ConfigStore.Delete = function(name)
    local path = ConfigStore.Path(name)
    local deleted = false
    if type(delfile) == "function" then
        deleted = pcall(delfile, path)
    elseif type(removefile) == "function" then
        deleted = pcall(removefile, path)
    end
    return deleted
end

ConfigStore.Prepare = function(cfg)
    if type(cfg) ~= "table" then return nil end

    local out = {
        Keybinds = {},
        Sliders = {},
        Features = {},
        AimTargetPart = cfg.AimTargetPart,
        TriggerbotTargetPart = cfg.TriggerbotTargetPart,
        ProtectedPlayers = {},
    }

    for k, v in pairs(cfg.Keybinds or {}) do
        if typeof(v) == "EnumItem" then
            out.Keybinds[k] = v.Name
        elseif type(v) == "string" then
            out.Keybinds[k] = v
        end
    end

    for k, v in pairs(cfg.Sliders or {}) do
        if type(v) == "number" then
            out.Sliders[k] = v
        end
    end

    for k, v in pairs(cfg.Features or {}) do
        if type(v) == "boolean" then
            out.Features[k] = v
        end
    end

    for userId, enabled in pairs(cfg.ProtectedPlayers or {}) do
        if enabled then
            out.ProtectedPlayers[tostring(userId)] = true
        end
    end

    return out
end

ConfigStore.Restore = function(data)
    if type(data) ~= "table" then return nil end

    local cfg = {
        Keybinds = {},
        Sliders = {},
        Features = {},
        AimTargetPart = data.AimTargetPart,
        TriggerbotTargetPart = data.TriggerbotTargetPart,
        ProtectedPlayers = {},
    }

    for k, v in pairs(data.Keybinds or {}) do
        if type(v) == "string" and Enum.KeyCode[v] then
            cfg.Keybinds[k] = Enum.KeyCode[v]
        end
    end

    for k, v in pairs(data.Sliders or {}) do
        if type(v) == "number" then
            cfg.Sliders[k] = v
        end
    end

    for k, v in pairs(data.Features or {}) do
        if type(v) == "boolean" then
            cfg.Features[k] = v
        end
    end

    for userId, enabled in pairs(data.ProtectedPlayers or {}) do
        if enabled then
            cfg.ProtectedPlayers[tonumber(userId) or userId] = true
        end
    end

    return cfg
end

ConfigStore.Save = function(name, cfg)
    if not ConfigStore.Ensure() then
        return false, "Executor file APIs are unavailable."
    end

    local payload = ConfigStore.Prepare(cfg)
    if not payload then
        return false, "Invalid config."
    end

    local ok, encoded = pcall(function()
        return game:GetService("HttpService"):JSONEncode(payload)
    end)
    if not ok then
        return false, "Config encoding failed."
    end

    local writeOk = pcall(function()
        writefile(ConfigStore.Path(name), encoded)
    end)
    if not writeOk then
        return false, "Could not write the config file."
    end

    return true, ConfigStore.Path(name)
end

ConfigStore.Load = function(name)
    if not ConfigStore.Ensure() or type(isfile) ~= "function" then
        return nil, "Executor file APIs are unavailable."
    end

    local path = ConfigStore.Path(name)
    if not isfile(path) then
        return nil, "Config file does not exist."
    end

    local readOk, raw = pcall(function()
        return readfile(path)
    end)
    if not readOk or type(raw) ~= "string" or raw == "" then
        return nil, "Could not read the config file."
    end

    local decodeOk, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(raw)
    end)
    if not decodeOk then
        return nil, "Config file is invalid."
    end

    return ConfigStore.Restore(data), path
end

local function CaptureConfig()
    local cfg = { Keybinds = {}, Sliders = {}, Features = {}, AimTargetPart = State.AimTargetPart, TriggerbotTargetPart = State.TriggerbotTargetPart, ProtectedPlayers = {} }
    for userId, enabled in pairs(State.ProtectedPlayers) do
        cfg.ProtectedPlayers[userId] = enabled
    end
    for k, v in pairs(CONFIG.Keybinds) do cfg.Keybinds[k] = v end
    for k, v in pairs(State.Sliders) do cfg.Sliders[k] = v end
    for k, v in pairs(State.Features) do
        if k ~= "AutoRememberKey" and k ~= "AutoLoadConfig" then
            cfg.Features[k] = v
        end
    end
    return cfg
end

local function ApplyConfig(cfg)
    if not cfg then return end
    for k, v in pairs(cfg.Keybinds or {}) do
        CONFIG.Keybinds[k] = v
    end
    for k, v in pairs(cfg.Sliders or {}) do
        State.Sliders[k] = v
    end
    for k, v in pairs(cfg.Features or {}) do
        State.Features[k] = v
    end
    if cfg.AimTargetPart then State.AimTargetPart = cfg.AimTargetPart end
    if cfg.TriggerbotTargetPart then State.TriggerbotTargetPart = cfg.TriggerbotTargetPart end
    State.ProtectedPlayers = {}
    for userId, enabled in pairs(cfg.ProtectedPlayers or {}) do
        if enabled then State.ProtectedPlayers[tonumber(userId) or userId] = true end
    end
    RefreshTracking()
    RefreshSelectedProtectionUI()
    RefreshPlayerListUI()
    ApplyESPColors()

RunService:BindToRenderStep("CheatMenuAimAssist", Enum.RenderPriority.Camera.Value + 1, function()
    Camera = Workspace.CurrentCamera or Camera
    if not State.Features.Aimbot or not Camera then return end

    local activationHeld = State.Features.AimbotAlwaysOn
    if not activationHeld then
        local keyHeld = State.Input.AimbotKey
        local mouseHeld = State.Input.RMB
        pcall(function()
            keyHeld = keyHeld or UserInputService:IsKeyDown(CONFIG.Keybinds.Aimbot or Enum.KeyCode.LeftAlt)
        end)
        pcall(function()
            mouseHeld = mouseHeld or UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
        end)
        if State.Features.AimbotUseCustomKey then
            activationHeld = keyHeld
        else
            activationHeld = mouseHeld or keyHeld
        end
    end
    if not activationHeld then return end

    if State.Sliders.AimbotHitChance < 100 and math.random(1, 100) > State.Sliders.AimbotHitChance then
        return
    end

    local viewportCenter = Camera.ViewportSize / 2
    local nearestPart, nearestScreenDist = nil, math.huge
    local nearestCharacter

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            local targetPart = GetAimTargetPart(char)
            local protected = IsProtectedPlayer(plr)
            local sameTeam = IsSameTeam(plr)
            if humanoid and humanoid.Health > 0 and targetPart and not protected and (not State.Features.TeamCheck or not sameTeam) and AimTargetVisible(targetPart) then
                local localRoot = char:FindFirstChild("HumanoidRootPart")
                local distance = localRoot and (localRoot.Position - Camera.CFrame.Position).Magnitude or math.huge
                if distance <= State.Sliders.AimbotMaxDistance then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                    if onScreen and screenPos.Z > 0 then
                        local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - viewportCenter).Magnitude
                        if screenDist <= State.Sliders.AimFOV and screenDist < nearestScreenDist then
                            nearestPart = targetPart
                            nearestScreenDist = screenDist
                            nearestCharacter = char
                        end
                    end
                end
            end
        end
    end

    if nearestPart then
        local aimPosition = nearestPart.Position
        if State.Features.AimbotPrediction then
            local velocity = nearestPart.AssemblyLinearVelocity
            aimPosition = aimPosition + velocity * State.Sliders.AimbotPredictionAmount
        end
        local targetCFrame = CFrame.new(Camera.CFrame.Position, aimPosition)
        Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, State.Sliders.AimSmoothness)
    end
end)

local function GetSilentAimTarget()
    Camera = Workspace.CurrentCamera or Camera
    if not Camera then return nil end

    local mousePos = Camera.ViewportSize / 2
    local bestPart, bestDistance = nil, math.huge

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            local part = GetAimTargetPart(char)
            local sameTeam = IsSameTeam(plr)
            local protected = IsProtectedPlayer(plr)
            if humanoid and humanoid.Health > 0 and part and not protected
                and (not State.Features.SilentAimTeamCheck or not sameTeam)
                and (not State.Features.SilentAimWallCheck or AimTargetVisible(part, true)) then

                local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and screenPos.Z > 0 then
                    local distance = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if distance <= State.Sliders.SilentAimFOV and distance < bestDistance then
                        bestPart = part
                        bestDistance = distance
                    end
                end
            end
        end
    end

    return bestPart
end

State.SilentAimHook = State.SilentAimHook or {Installed = false, Old = nil}

local function IsSilentAimRemote(remote)
    if typeof(remote) ~= "Instance" then return false end
    if not (remote:IsA("RemoteEvent") or remote.ClassName == "UnreliableRemoteEvent") then return false end
    local name = remote.Name:lower()
    for _, token in ipairs({"shoot","fire","attack","hit","damage","bullet","weapon","gun","ray","projectile"}) do
        if name:find(token, 1, true) then return true end
    end
    return false
end

local function InstallSilentAimHook()
    if State.SilentAimHook.Installed then return end
    if type(hookmetamethod) ~= "function" or type(newcclosure) ~= "function" or type(getnamecallmethod) ~= "function" then
        return
    end

    local old
    local wrapper = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local callerIsScript = false
        if type(checkcaller) == "function" then
            callerIsScript = checkcaller()
        end

        if State.Features.SilentAim and not callerIsScript and (method == "FireServer" or method == "InvokeServer") and IsSilentAimRemote(self) then
            local target = GetSilentAimTarget()
            if target and State.Sliders.SilentAimHitChance >= 100 or (target and math.random(1, 100) <= State.Sliders.SilentAimHitChance) then
                local targetPosition = target.Position
                if State.Features.SilentAimPrediction then
                    targetPosition = targetPosition + target.AssemblyLinearVelocity * State.Sliders.SilentAimPredictionAmount
                end

                local args = {...}
                local changed = false
                for i, value in ipairs(args) do
                    if typeof(value) == "Vector3" then
                        args[i] = targetPosition
                        changed = true
                    elseif typeof(value) == "Ray" then
                        args[i] = Ray.new(value.Origin, targetPosition - value.Origin)
                        changed = true
                    elseif typeof(value) == "CFrame" then
                        args[i] = CFrame.new(value.Position, targetPosition)
                        changed = true
                    end
                end

                if changed then
                    return old(self, table.unpack(args))
                end
            end
        end

        return old(self, ...)
    end)

    local ok, previous = pcall(function()
        return hookmetamethod(game, "__namecall", wrapper)
    end)
    if ok and previous then
        State.SilentAimHook.Old = previous
        State.SilentAimHook.Installed = true
    end
end

local function RestoreSilentAimHook()
    if not State.SilentAimHook.Installed or type(hookmetamethod) ~= "function" or not State.SilentAimHook.Old then return end
    pcall(function()
        hookmetamethod(game, "__namecall", State.SilentAimHook.Old)
    end)
    State.SilentAimHook.Installed = false
    State.SilentAimHook.Old = nil
end

    pcall(function() triggerKeyHeld = triggerKeyHeld or UserInputService:IsKeyDown(CONFIG.Keybinds.Triggerbot or Enum.KeyCode.LeftAlt) end)
    if State.Features.TriggerbotOnlyWhileMouse1 and not lmbHeld then return end
    if not State.Features.TriggerbotAlwaysOn and not triggerKeyHeld then return end

    if os.clock() - triggerLastClick < State.Sliders.TriggerbotCooldown then return end

    local viewportMouse
    if State.Features.TriggerbotUseCrosshair then
        viewportMouse = Camera.ViewportSize / 2
    else
        viewportMouse = UserInputService:GetMouseLocation() - game:GetService("GuiService"):GetGuiInset()
    end
    local bestPart, bestDistance, bestPlayer = nil, math.huge, nil

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            local targetPart = GetAimTargetPart(char, State.TriggerbotTargetPart)
            local protected = IsProtectedPlayer(plr)
            local sameTeam = LocalPlayer.Team ~= nil and plr.Team ~= nil and LocalPlayer.Team == plr.Team
            if humanoid and humanoid.Health > 0 and targetPart and not protected and (not State.Features.TriggerbotTeamCheck or not sameTeam) then
                if not State.Features.TriggerbotWallCheck or AimTargetVisible(targetPart, true) then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                    if onScreen and screenPos.Z > 0 then
                        local screenDistance = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(viewportMouse.X, viewportMouse.Y)).Magnitude
                        if screenDistance <= State.Sliders.TriggerbotFOV and screenDistance < bestDistance then
                            bestPart = targetPart
                            bestDistance = screenDistance
                            bestPlayer = plr
                        end
                    end
                end
            end
        end
    end

    if not bestPart or triggerPending then
        triggerTargetPlayer = bestPlayer
        return
    end

    if State.Sliders.TriggerbotHitChance < 100 and math.random(1, 100) > State.Sliders.TriggerbotHitChance then
        return
    end

    triggerPending = true
    triggerTargetPlayer = bestPlayer
    local expectedTarget = bestPlayer
    task.delay(State.Sliders.TriggerbotDelay, function()
        triggerPending = false
        if not State.Features.Triggerbot or not expectedTarget or expectedTarget ~= triggerTargetPlayer then return end
        triggerLastClick = os.clock()
        PerformTriggerClick()
    end)
end)

local fpsAccum, fpsFrames = 0, 0

RunService.Heartbeat:Connect(function(dt)

    fpsAccum += dt
    fpsFrames += 1
    if fpsAccum >= 0.5 then
        UI.FPSLabel.Text = string.format("FPS: %d", math.floor(fpsFrames / fpsAccum + 0.5))
        fpsAccum, fpsFrames = 0, 0
    end

    if not Camera or not Camera.Parent then
        Camera = Workspace.CurrentCamera
    end
    if not Camera then return end

    if State.UpdateRadar then State.UpdateRadar() end

    if State.Features.ShowFOV then
        local vp = Camera.ViewportSize
        local fov = State.Sliders.AimFOV
        FOVCircle.Visible = true
        FOVCircle.Position = UDim2.fromOffset(vp.X / 2, vp.Y / 2)
        FOVCircle.Size = UDim2.fromOffset(fov * 2, fov * 2)
    else
        FOVCircle.Visible = false
    end

    if FreecamStatusLabel then
        if State.Features.Freecam and FreecamPosition and not State.Features.HideFreecamBar then
            FreecamStatusLabel.Visible = true
            FreecamStatusLabel.Text = string.format("FREECAM  X %.1f  Y %.1f  Z %.1f   |   RightCtrl = TELEPORT", FreecamPosition.X, FreecamPosition.Y, FreecamPosition.Z)
        else
            FreecamStatusLabel.Visible = false
        end
    end

    if State.Features.ToolViewer then
        RefreshToolViewer()
    end

    if State.Features.InventoryView then
        State.UpdateInventoryView()
    end

    if ShouldTrack() then
        local camPos = Camera.CFrame.Position
        local viewportSize = Camera.ViewportSize
        for plr, objs in pairs(State.ESPObjects) do
            local protected = IsProtectedPlayer(plr)
            local char = objs.Character
            if char ~= plr.Character then
                RemoveVisualsOnly(plr)
                char = nil
            end
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local dist = (hrp.Position - camPos).Magnitude
                local withinRange = dist <= State.Sliders.ESPDistance

                local minX, minY, maxX, maxY = GetCharacterScreenBounds(char, Camera)
                local boundsVisible = withinRange and minX ~= nil
                if objs.BoxFrame then
                    objs.BoxFrame.Visible = State.Features.BoxESP and boundsVisible
                    if boundsVisible and State.Features.BoxESP then
                        objs.BoxFrame.Position = UDim2.fromOffset(minX, minY)
                        objs.BoxFrame.Size = UDim2.fromOffset(math.max(1, maxX - minX), math.max(1, maxY - minY))
                        objs.BoxStroke.Color = protected and CONFIG.Colors.ProtectedESP or CONFIG.Colors.BoxESP
                    end
                end
                if objs.HealthBar then
                    objs.HealthBar.Visible = State.Features.Healthbar and boundsVisible
                    if boundsVisible and State.Features.Healthbar then
                        local h = char:FindFirstChildOfClass("Humanoid")
                        local hp = h and math.max(0, h.Health) or 0
                        local maxHp = h and math.max(1, h.MaxHealth) or 1
                        local fraction = math.clamp(hp / maxHp, 0, 1)
                        local barHeight = math.max(12, maxY - minY)
                        objs.HealthBar.Position = UDim2.fromOffset(math.max(0, minX - 7), minY)
                        objs.HealthBar.Size = UDim2.fromOffset(4, barHeight)
                        objs.HealthFill.Size = UDim2.new(1, 0, fraction, 0)
                        objs.HealthFill.BackgroundColor3 = fraction > 0.5 and CONFIG.Colors.HealthGood or CONFIG.Colors.HealthLow
                    end
                end

                local highlightFill, highlightOutline = State.GetHighlightColors(plr)
                if objs.Highlight then
                    objs.Highlight.FillColor = highlightFill
                    objs.Highlight.OutlineColor = highlightOutline
                    objs.Highlight.DepthMode = State.Features.Chams and Enum.HighlightDepthMode.AlwaysOnTop or Enum.HighlightDepthMode.Occluded
                end

                if State.Features.ESP then
                    objs.Highlight.Enabled = withinRange or State.Features.Chams
                    if objs.NameLabel then objs.NameLabel.TextColor3 = protected and CONFIG.Colors.ProtectedESP or Color3.fromRGB(255,255,255) end
                    objs.Billboard.Enabled = withinRange
                    objs.Billboard.MaxDistance = State.Sliders.ESPDistance
                    objs.DistLabel.Text = string.format("%d studs", dist)
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if objs.HealthLabel and hum then
                        local hp = math.max(0, hum.Health)
                        local maxHp = math.max(1, hum.MaxHealth)
                        objs.HealthLabel.Text = string.format("HP %d / %d", math.floor(hp + 0.5), math.floor(maxHp + 0.5))
                    end
                else
                    objs.Billboard.Enabled = false
                end

                if State.Features.SkeletonESP and ((#(objs.SkeletonSegments or {}) == 0) or not objs.SkeletonHead or not objs.SkeletonHead.Parent) and char.Parent then
                    RebuildSkeleton(plr)
                end

                if State.Features.SkeletonESP and State.Features.SkeletonHeadMarker and withinRange and objs.SkeletonHead then
                    local headPart = char:FindFirstChild("Head")
                    if headPart and headPart:IsA("BasePart") then
                        local sh, headOnScreen = Camera:WorldToViewportPoint(headPart.Position + Vector3.new(0, 0.15, 0))
                        if headOnScreen then
                            objs.SkeletonHead.Position = UDim2.fromOffset(sh.X, sh.Y)

                            local mousePos = UserInputService:GetMouseLocation() - GuiService:GetGuiInset()
                            local headScreen = Vector2.new(sh.X, sh.Y)
                            local cursorDistance = (headScreen - mousePos).Magnitude
                            local headAligned = State.Features.HeadTargetIndicator and cursorDistance <= 14

                            if headAligned then
                                objs.SkeletonHead.Size = UDim2.fromOffset(State.Sliders.SkeletonHeadTargetSize, State.Sliders.SkeletonHeadTargetSize)
                                objs.SkeletonHead.BackgroundColor3 = CONFIG.Colors.HeadTarget
                            else
                                objs.SkeletonHead.Size = UDim2.fromOffset(State.Sliders.SkeletonHeadSize, State.Sliders.SkeletonHeadSize)
                                if protected then
                                    objs.SkeletonHead.BackgroundColor3 = CONFIG.Colors.ProtectedESP
                                elseif State.Features.SkeletonVisibilityCheck then
                                    objs.SkeletonHead.BackgroundColor3 = AimTargetVisible(headPart, true) and CONFIG.Colors.SkeletonVisible or CONFIG.Colors.SkeletonHidden
                                else
                                    objs.SkeletonHead.BackgroundColor3 = CONFIG.Colors.Skeleton
                                end
                            end

                            objs.SkeletonHead.Visible = true
                        else
                            objs.SkeletonHead.Visible = false
                        end
                    else
                        objs.SkeletonHead.Visible = false
                    end
                elseif objs.SkeletonHead then
                    objs.SkeletonHead.Visible = false
                end

                for _, segment in ipairs(objs.SkeletonSegments or {}) do
                    local frame = segment.Frame
                    local glow = segment.Glow
                    local a = char:FindFirstChild(segment.A)
                    local b = char:FindFirstChild(segment.B)
                    if frame and a and b and a:IsA("BasePart") and b:IsA("BasePart") and State.Features.SkeletonESP and withinRange then
                        local sa, ona = Camera:WorldToViewportPoint(a.Position)
                        local sb, onb = Camera:WorldToViewportPoint(b.Position)
                        if ona and onb and sa.Z > 0 and sb.Z > 0 then
                            local dx = sb.X - sa.X
                            local dy = sb.Y - sa.Y
                            local length = math.sqrt(dx*dx + dy*dy)
                            if length > 1 then
                                local midX, midY = (sa.X + sb.X) * 0.5, (sa.Y + sb.Y) * 0.5
                                local rotation = math.deg(math.atan2(-dx, dy))
                                if protected then
                                    frame.BackgroundColor3 = CONFIG.Colors.ProtectedESP
                                elseif State.Features.SkeletonVisibilityCheck then
                                    frame.BackgroundColor3 = AimTargetVisible(b, true) and CONFIG.Colors.SkeletonVisible or CONFIG.Colors.SkeletonHidden
                                else
                                    frame.BackgroundColor3 = CONFIG.Colors.Skeleton
                                end
                                if glow then
                                    glow.BackgroundColor3 = frame.BackgroundColor3
                                    glow.Visible = State.Features.SkeletonGlow
                                end
                                frame.Size = UDim2.fromOffset(math.max(1, State.Sliders.SkeletonThickness), length)
                                frame.Position = UDim2.fromOffset(midX, midY)
                                frame.Rotation = rotation
                                frame.Visible = true
                                if glow then
                                    glow.Size = UDim2.fromOffset(math.max(1, State.Sliders.SkeletonGlowThickness), length)
                                    glow.Position = UDim2.fromOffset(midX, midY)
                                    glow.Rotation = rotation
                                    glow.Visible = true
                                end
                            else
                                frame.Visible = false
                                if glow then glow.Visible = false end
                            end
                        else
                            frame.Visible = false
                            if glow then glow.Visible = false end
                        end
                    elseif frame then
                        frame.Visible = false
                        if glow then glow.Visible = false end
                    end
                end

                for _, joint in ipairs(objs.SkeletonJoints or {}) do
                    local part = char:FindFirstChild(joint.PartName)
                    local dot = joint.Frame
                    if dot and part and part:IsA("BasePart") and State.Features.SkeletonESP and State.Features.SkeletonJoints and withinRange then
                        local sj, onScreen = Camera:WorldToViewportPoint(part.Position)
                        if onScreen and sj.Z > 0 then
                            dot.Position = UDim2.fromOffset(sj.X, sj.Y)
                            if protected then
                                dot.BackgroundColor3 = CONFIG.Colors.ProtectedESP
                            elseif State.Features.SkeletonVisibilityCheck then
                                dot.BackgroundColor3 = AimTargetVisible(part, true) and CONFIG.Colors.SkeletonVisible or CONFIG.Colors.SkeletonHidden
                            else
                                dot.BackgroundColor3 = CONFIG.Colors.Skeleton
                            end
                            dot.Visible = true
                        else
                            dot.Visible = false
                        end
                    elseif dot then
                        dot.Visible = false
                    end
                end

                if State.Features.Tracers and withinRange then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        local originX, originY = viewportSize.X / 2, viewportSize.Y
                        local dx, dy = screenPos.X - originX, screenPos.Y - originY
                        local length = math.sqrt(dx * dx + dy * dy)
                        local angle = math.deg(math.atan2(dy, dx))
                        objs.Tracer.Visible = true
                        objs.Tracer.BackgroundColor3 = protected and CONFIG.Colors.ProtectedESP or CONFIG.Colors.Tracer
                        objs.Tracer.Size = UDim2.fromOffset(length, 2)
                        objs.Tracer.Position = UDim2.fromOffset((originX + screenPos.X) / 2, (originY + screenPos.Y) / 2)
                        objs.Tracer.Rotation = angle
                    else
                        objs.Tracer.Visible = false
                    end
                else
                    objs.Tracer.Visible = false
                end
            end
        end
    end
end)
