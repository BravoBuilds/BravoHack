local function Tween(obj, info, props)
    local tw = TweenService:Create(obj, info, props)
    tw:Play()
    return tw

local function NewCorner(radius, parent)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = parent
    return c
end

local function NewStroke(parent, color, thickness, transparency)

    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = parent

local function RefreshTabVisuals()
    for tabName, tabButton in pairs(SidebarButtons) do
        local selected = (tabName == State.CurrentTab)
        Tween(tabButton, TI_FAST, {
            TextColor3 = selected and CONFIG.Colors.Text or CONFIG.Colors.SubText,
            BackgroundTransparency = selected and 0.82 or 1,
        })
    end
    local currentButton = SidebarButtons[State.CurrentTab]
    if currentButton then
        Tween(Indicator, TI_MED, {
            Position = UDim2.new(0, 0, 0, currentButton.AbsolutePosition.Y - Sidebar.AbsolutePosition.Y),
        })
    end
end

local function SwitchTab(name)
    if not Pages[name] or not SidebarButtons[name] then return end
    local previousPage = Pages[State.CurrentTab]
    local nextPage = Pages[name]
    if previousPage == nextPage then
        RefreshTabVisuals()
        return
    end

    previousPage.Visible = false
    State.CurrentTab = name
    TabTitle.Text = name
    local basePos = PageBasePositions[name] or UDim2.new(0, 10, 0, 54)
    nextPage.Position = UDim2.new(basePos.X.Scale, basePos.X.Offset + 14, basePos.Y.Scale, basePos.Y.Offset)
    nextPage.Visible = true
    Tween(nextPage, TI_MED, { Position = basePos })
    RefreshTabVisuals()
end

for name, btn in pairs(SidebarButtons) do
    btn.MouseEnter:Connect(function()
        Tween(btn, TI_FAST, { TextColor3 = CONFIG.Colors.Text, BackgroundTransparency = 0.88 })
    end)
    btn.MouseLeave:Connect(function()
        local selected = (State.CurrentTab == name)
        Tween(btn, TI_FAST, {
            TextColor3 = selected and CONFIG.Colors.Text or CONFIG.Colors.SubText,
            BackgroundTransparency = selected and 0.82 or 1,
        })
    end)
    btn.MouseButton1Click:Connect(function()
        SwitchTab(name)
    end)
end

task.defer(function()
    RefreshTabVisuals()
end)

local function TriggerKeybind(keyCode)
    if not keyCode then return false end
    for _, entry in pairs(FeatureKeybindRefs) do
        local bound = entry.Get and entry.Get()
        if bound and bound == keyCode and entry.Toggle then
            entry.Toggle()
            return true
        end
    end
    for _, entry in pairs(ActionKeybindRefs) do
        local bound = entry.Get and entry.Get()
        if bound and bound == keyCode and entry.Trigger then
            entry.Trigger()
            return true
        end
    end
    return false
end

local function GetCharacterScreenBounds(char, camera)
    if not char or not camera then return nil end
    local ok, cf, size = pcall(function() return char:GetBoundingBox() end)
    if not ok or not cf or not size then return nil end
    local half = size * 0.5
    local corners = {
        Vector3.new(-half.X, -half.Y, -half.Z), Vector3.new(-half.X, -half.Y, half.Z),
        Vector3.new(-half.X, half.Y, -half.Z), Vector3.new(-half.X, half.Y, half.Z),
        Vector3.new(half.X, -half.Y, -half.Z), Vector3.new(half.X, -half.Y, half.Z),
        Vector3.new(half.X, half.Y, -half.Z), Vector3.new(half.X, half.Y, half.Z),
    }
    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local any = false
    for _, offset in ipairs(corners) do
        local screen, onScreen = camera:WorldToViewportPoint(cf:PointToWorldSpace(offset))
        if screen.Z > 0 then
            any = true
            minX = math.min(minX, screen.X)
            minY = math.min(minY, screen.Y)
            maxX = math.max(maxX, screen.X)
            maxY = math.max(maxY, screen.Y)
        end
    end
    if not any then return nil end
    local vp = camera.ViewportSize
    minX = math.clamp(minX, 0, vp.X)
    minY = math.clamp(minY, 0, vp.Y)
    maxX = math.clamp(maxX, 0, vp.X)
    maxY = math.clamp(maxY, 0, vp.Y)
    if maxX - minX < 4 or maxY - minY < 8 then return nil end
    return minX, minY, maxX, maxY
end

function RefreshTracking()
    if ShouldTrack() then
        for _, plr in ipairs(Players:GetPlayers()) do
            AddESPForPlayer(plr)
        end
    else
        local toRemove = {}
        for plr in pairs(State.ESPObjects) do
            table.insert(toRemove, plr)
        end
        for _, plr in ipairs(toRemove) do
            RemoveESPForPlayer(plr)
        end
    end
end

local FlyConn, FlyBodyVelocity
function EnableFly()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    FlyBodyVelocity = Instance.new("BodyVelocity")
    FlyBodyVelocity.Name = "CheatFlyForce"
    FlyBodyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    FlyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
    FlyBodyVelocity.Parent = hrp

    if FlyConn then FlyConn:Disconnect() end
    FlyConn = RunService.Heartbeat:Connect(function()
        if not State.Features.Fly or not FlyBodyVelocity or not FlyBodyVelocity.Parent then return end
        local cf = Camera.CFrame
        local dir = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cf.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cf.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cf.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cf.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.new(0, 1, 0) end
        if dir.Magnitude > 0 then dir = dir.Unit end
        FlyBodyVelocity.Velocity = dir * math.max(State.Sliders.WalkSpeed * 3, 20)
    end)
end
function DisableFly()
    if FlyConn then
        FlyConn:Disconnect()
        FlyConn = nil
    end
    if FlyBodyVelocity then
        FlyBodyVelocity:Destroy()
        FlyBodyVelocity = nil
    end
end

local FreecamConn
local SavedCameraType, SavedCameraSubject
local SavedMouseBehavior, SavedMouseIconEnabled

local FreecamSavedRootCFrame = nil
local FreecamSavedRootAnchored = nil
local FreecamSavedAutoRotate = nil
local FreecamSavedWalkSpeed = nil
local FreecamSavedJumpPower = nil
local FreecamChatFocused = false

local function FreezeCharacterForFreecam(forceCapture)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not hrp then return false end

    if forceCapture or FreecamSavedRootCFrame == nil then FreecamSavedRootCFrame = hrp.CFrame end
    if FreecamSavedRootAnchored == nil then FreecamSavedRootAnchored = hrp.Anchored end
    if hum then
        if FreecamSavedAutoRotate == nil then FreecamSavedAutoRotate = hum.AutoRotate end
        if FreecamSavedWalkSpeed == nil then FreecamSavedWalkSpeed = hum.WalkSpeed end
        if FreecamSavedJumpPower == nil then FreecamSavedJumpPower = hum.JumpPower end
        hum.AutoRotate = false
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum:ChangeState(Enum.HumanoidStateType.Physics)
    end
    hrp.Anchored = true
    hrp.CFrame = FreecamSavedRootCFrame
    return true
end

local function RestoreCharacterAfterFreecam()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
        if FreecamSavedRootCFrame then hrp.CFrame = FreecamSavedRootCFrame end
        if FreecamSavedRootAnchored ~= nil then hrp.Anchored = FreecamSavedRootAnchored end
    end
    if hum then
        if FreecamSavedAutoRotate ~= nil then hum.AutoRotate = FreecamSavedAutoRotate end
        if FreecamSavedWalkSpeed ~= nil then hum.WalkSpeed = FreecamSavedWalkSpeed end
        if FreecamSavedJumpPower ~= nil then hum.JumpPower = FreecamSavedJumpPower end
        if hum.Health > 0 then hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end
    FreecamSavedRootCFrame = nil
    FreecamSavedRootAnchored = nil
    FreecamSavedAutoRotate = nil
    FreecamSavedWalkSpeed = nil
    FreecamSavedJumpPower = nil
end

    Camera.Focus = CFrame.new(FreecamPosition + rotation.LookVector * 100)
end

    if chatFocused then

local function FocusRobloxChat()
    ChatInputBarConfiguration = TextChatService:FindFirstChild("ChatInputBarConfiguration") or ChatInputBarConfiguration
    if ChatInputBarConfiguration and ChatInputBarConfiguration.Enabled then
        local textBox = ChatInputBarConfiguration.TextBox
        if textBox and textBox.TextEditable then
            FreecamChatFocused = true
            SetFreecamInputMode(true)
            pcall(function() textBox:CaptureFocus() end)
            return true
        end
    end

    for _, obj in ipairs(PlayerGui:GetDescendants()) do
        if obj:IsA("TextBox") and obj.Visible and obj.TextEditable then
            FreecamChatFocused = true
            SetFreecamInputMode(true)
            pcall(function() obj:CaptureFocus() end)
            return true
        end
    end
    return false
end

function EnableFreecam()
    Camera = Workspace.CurrentCamera or Camera
    if not Camera then return end
    SavedCameraType = Camera.CameraType
    SavedCameraSubject = Camera.CameraSubject
    SavedMouseBehavior = UserInputService.MouseBehavior
    SavedMouseIconEnabled = UserInputService.MouseIconEnabled

    local look = Camera.CFrame.LookVector
    FreecamPosition = Camera.CFrame.Position
    FreecamYaw = math.atan2(-look.X, -look.Z)
    FreecamPitch = math.clamp(math.asin(math.clamp(look.Y, -0.999, 0.999)), math.rad(-85), math.rad(85))

    FreezeCharacterForFreecam()
    Camera.CameraType = Enum.CameraType.Scriptable
    Camera.FieldOfView = State.Sliders.CameraFOV

    FreecamChatFocused = UserInputService:GetFocusedTextBox() ~= nil
    SetFreecamInputMode(FreecamChatFocused)
    SetFreecamCamera()

    if FreecamConn then FreecamConn:Disconnect() end
    RunService:BindToRenderStep("DeveloperFreecam", Enum.RenderPriority.Camera.Value + 5, function(dt)
        if not State.Features.Freecam then return end
        Camera = Workspace.CurrentCamera or Camera
        if not Camera then return end
        FreezeCharacterForFreecam(false)

        local focusedBox = UserInputService:GetFocusedTextBox()
        local chatFocused = FreecamChatFocused or focusedBox ~= nil
        if ChatInputBarConfiguration and ChatInputBarConfiguration.IsFocused then
            chatFocused = true
        end
        if chatFocused then
            FreecamChatFocused = true
            SetFreecamInputMode(true)
            Camera.CameraType = Enum.CameraType.Scriptable
            return
        end
        SetFreecamInputMode(false)

        local delta = UserInputService:GetMouseDelta()
        FreecamYaw -= delta.X * 0.0028
        FreecamPitch = math.clamp(FreecamPitch - delta.Y * 0.0028, math.rad(-85), math.rad(85))

        local rotation = CFrame.fromOrientation(FreecamPitch, FreecamYaw, 0)
        local move = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += rotation.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= rotation.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= rotation.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += rotation.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.yAxis end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.yAxis end
        if move.Magnitude > 0 then
            local speed = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) and FreecamSpeed * 3 or FreecamSpeed
            FreecamPosition += move.Unit * speed * dt
        end
        Camera.CameraType = Enum.CameraType.Scriptable
        Camera.CFrame = CFrame.new(FreecamPosition) * rotation
        Camera.Focus = CFrame.new(FreecamPosition + rotation.LookVector * 100)
    end)
end

function DisableFreecam()

    pcall(function() RunService:UnbindFromRenderStep("DeveloperFreecam") end)
    if FreecamConn then
        FreecamConn:Disconnect()
        FreecamConn = nil
    end

    pcall(function()
        local focused = UserInputService:GetFocusedTextBox()
        if focused then focused:ReleaseFocus() end
    end)

    if Camera then
        Camera.CameraType = SavedCameraType or Enum.CameraType.Custom
        if SavedCameraSubject and SavedCameraSubject.Parent then
            Camera.CameraSubject = SavedCameraSubject
        else
            local hum = GetHumanoid()
            if hum then Camera.CameraSubject = hum end
        end
    end

    FreecamChatFocused = false

    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    UserInputService.MouseIconEnabled = true

    RestoreCharacterAfterFreecam()
    FreecamPosition = nil

    task.defer(function()
        if not State.Features.Freecam then
            UserInputService.MouseBehavior = Enum.MouseBehavior.Default
            UserInputService.MouseIconEnabled = true
        end
    end)
end

local MiscUI = {}

local function GetPlayerTools(plr)
    local names, seen = {}, {}
    if not plr then return names end

    local backpack = plr:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, obj in ipairs(backpack:GetChildren()) do
            if obj:IsA("Tool") and not seen[obj.Name] then
                seen[obj.Name] = true
                table.insert(names, obj.Name)
            end
        end
    end

    local char = plr.Character
    if char then
        for _, obj in ipairs(char:GetChildren()) do
            if obj:IsA("Tool") and not seen[obj.Name] then
                seen[obj.Name] = true
                table.insert(names, obj.Name .. " [equipped]")
            end
        end
    end

    table.sort(names)
    return names
end

local function FindCrosshairPlayer()
    if not Camera then return nil end
    local vp = Camera.ViewportSize
    local ray = Camera:ViewportPointToRay(vp.X * 0.5, vp.Y * 0.5)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character}

    local result = Workspace:Raycast(ray.Origin, ray.Direction * 2000, params)
    if not result or not result.Instance then return nil end
    local model = result.Instance:FindFirstAncestorOfClass("Model")
    return model and Players:GetPlayerFromCharacter(model) or nil
end

local SpectateTarget = nil
local SpectateSavedCameraType = nil
local SpectateSavedCameraSubject = nil
local SpectateCharacterConn = nil

local function ApplySpectateTarget(plr)
    if not plr or plr == LocalPlayer then return false end
    Camera = Workspace.CurrentCamera or Camera
    if not Camera then return false end

    local char = plr.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then
        return false
    end

    Camera.CameraType = Enum.CameraType.Custom
    Camera.CameraSubject = hum
    return true
end

local function StopSpectatingSelectedPlayer()
    State.Features.Spectate = false

    if SpectateCharacterConn then
        SpectateCharacterConn:Disconnect()
        SpectateCharacterConn = nil
    end

    Camera = Workspace.CurrentCamera or Camera
    if Camera then
        Camera.CameraType = SpectateSavedCameraType or Enum.CameraType.Custom
        if SpectateSavedCameraSubject and SpectateSavedCameraSubject.Parent then
            Camera.CameraSubject = SpectateSavedCameraSubject
        else
            local hum = GetHumanoid()
            if hum then Camera.CameraSubject = hum end
        end
    end

    SpectateTarget = nil
    SpectateSavedCameraType = nil
    SpectateSavedCameraSubject = nil
end

local function StartSpectatingSelectedPlayer()
    local target = State.SelectedPlayer
    if not target or target == LocalPlayer or not target.Parent then
        State.Features.Spectate = false
        if Notify then Notify("Spectate", "Selecteer eerst een speler in de Playerlist.", true) end
        return false
    end

    if State.Features.Freecam then
        State.Features.Spectate = false
        if Notify then Notify("Spectate", "Zet Freecam eerst uit.", true) end
        return false
    end

    Camera = Workspace.CurrentCamera or Camera
    if not Camera then
        State.Features.Spectate = false
        return false
    end

    if not State.Features.Spectate then
        SpectateSavedCameraType = Camera.CameraType
        SpectateSavedCameraSubject = Camera.CameraSubject
    end

    State.Features.Spectate = true
    SpectateTarget = target

    if SpectateCharacterConn then
        SpectateCharacterConn:Disconnect()
        SpectateCharacterConn = nil
    end

    SpectateCharacterConn = target.CharacterAdded:Connect(function()
        if not State.Features.Spectate or SpectateTarget ~= target then return end
        task.wait(0.15)
        ApplySpectateTarget(target)
    end)

    if not ApplySpectateTarget(target) then
        StopSpectatingSelectedPlayer()
        if Notify then Notify("Spectate", "Deze speler heeft momenteel geen character.", true) end
        return false
    end

    if Notify then Notify("Spectate", "Nu kijk je mee met " .. target.Name .. ".", true) end
    return true
end

local function UpdateInventoryView()
    if not MiscUI.InventoryLabel then return end
    if not State.Features.InventoryView then
        return
    end

    local target = FindCrosshairPlayer()
    if not target then
        MiscUI.InventoryLabel.Text = "Crosshair: geen speler"
        return
    end

    local tools = GetPlayerTools(target)
    if target ~= LocalPlayer then
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\nZichtbare/equipped tools:" .. (#tools > 0 and ("\n" .. table.concat(tools, "\n")) or "\nGeen zichtbare equipped tools.")
    elseif #tools == 0 then
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\nGeen items gevonden."
    else
        MiscUI.InventoryLabel.Text = "@" .. target.Name .. "\n" .. table.concat(tools, "\n")
    end
end

local function RefreshToolViewer()

        if not State.Features.ToolViewer then
            RemoveToolViewerForPlayer(plr)
        end
    end
    if not State.Features.ToolViewer then return end

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local gui = ToolViewerObjects[plr]
            local char = plr.Character
            local head = char and char:FindFirstChild("Head")
            if not gui or not gui.Parent or not head then
                CreateToolViewerForPlayer(plr)
                gui = ToolViewerObjects[plr]
            end
            if gui and gui.Parent then
                local label = gui:FindFirstChildOfClass("TextLabel")
                if label then
                    local tools = GetPlayerTools(plr)
                    label.Text = plr.DisplayName .. "\nTools: " .. (#tools > 0 and table.concat(tools, ", ") or "geen lokaal zichtbare tools")
                    gui.MaxDistance = State.Sliders.ESPDistance
                end
            end
        end
    end
end

local MiscWindowSizes = {}
local MiscWindowBusy = {}

SetMiscWindowVisible = function(frame, visible)
    if not frame or not frame.Parent then return end

    local originalSize = MiscWindowSizes[frame]
    if not originalSize then
        originalSize = frame.Size
        MiscWindowSizes[frame] = originalSize
    end

    frame:SetAttribute("ShouldBeVisible", visible)
    if MiscWindowBusy[frame] then return end

    if visible then
        MiscWindowBusy[frame] = true
        frame.Visible = true
        frame.Size = UDim2.fromOffset(0, 0)
        local tw = Tween(frame, TI_SLOW, { Size = originalSize })
        tw.Completed:Connect(function()
            MiscWindowBusy[frame] = nil
            if frame.Parent and frame:GetAttribute("ShouldBeVisible") then
                frame.Size = originalSize
            end
        end)
    else
        MiscWindowBusy[frame] = true
        local tw = Tween(frame, TI_MED, { Size = UDim2.fromOffset(0, 0) })
        tw.Completed:Connect(function()
            MiscWindowBusy[frame] = nil
            if frame.Parent and not frame:GetAttribute("ShouldBeVisible") then
                frame.Visible = false
                frame.Size = originalSize
            end
        end)
    end
end

local function CaptureConfig()
    local cfg = { Keybinds = {}, Sliders = {}, Features = {}, AimTargetPart = State.AimTargetPart }

RunService:BindToRenderStep("CheatMenuAimAssist", Enum.RenderPriority.Camera.Value + 1, function()
    if not State.Features.AimAssist then return end
    if not Camera then return end
    if not UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end

    local viewportCenter = Camera.ViewportSize / 2
    local nearestPart, nearestScreenDist = nil, math.huge

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local targetPart = GetAimTargetPart(char)
            if targetPart then
                local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - viewportCenter).Magnitude
                    if screenDist <= State.Sliders.AimFOV and screenDist < nearestScreenDist then
                        nearestPart = targetPart
                        nearestScreenDist = screenDist
                    end
                end
            end
        end
    end

    if nearestPart then
        local targetCFrame = CFrame.new(Camera.CFrame.Position, nearestPart.Position)
        Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, State.Sliders.AimSmoothness)
    end
end)

local fpsAccum, fpsFrames = 0, 0

RunService.Heartbeat:Connect(function(dt)

    fpsAccum += dt
    fpsFrames += 1
    if fpsAccum >= 0.5 then
        FPSLabel.Text = string.format("FPS: %d", math.floor(fpsFrames / fpsAccum + 0.5))
        fpsAccum, fpsFrames = 0, 0
    end

    if not Camera or not Camera.Parent then
        Camera = Workspace.CurrentCamera
    end
    if not Camera then return end

    if State.Features.ShowFOV then
        local vp = Camera.ViewportSize
        FOVCircle.Visible = true
        FOVCircle.Position = UDim2.fromOffset(vp.X / 2, vp.Y / 2)
        FOVCircle.Size = UDim2.fromOffset(State.Sliders.AimFOV * 2, State.Sliders.AimFOV * 2)
    else
        FOVCircle.Visible = false
    end

    if FreecamStatusLabel then
        if State.Features.Freecam and FreecamPosition then
            FreecamStatusLabel.Visible = true
            FreecamStatusLabel.Text = string.format("FREECAM  X %.1f  Y %.1f  Z %.1f   |   RightCtrl = TELEPORT", FreecamPosition.X, FreecamPosition.Y, FreecamPosition.Z)
        else
            FreecamStatusLabel.Visible = false
        end
    end

    if State.Features.ToolViewer then
        RefreshToolViewer()
    end

    if State.Features.InventoryView then
        UpdateInventoryView()
    end

    if ShouldTrack() then
        local camPos = Camera.CFrame.Position
        local viewportSize = Camera.ViewportSize
        for plr, objs in pairs(State.ESPObjects) do
            local char = objs.Character
            if char ~= plr.Character then
                RemoveVisualsOnly(plr)
                char = nil
            end
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local dist = (hrp.Position - camPos).Magnitude
                local withinRange = dist <= State.Sliders.ESPDistance

                local minX, minY, maxX, maxY = GetCharacterScreenBounds(char, Camera)
                local boundsVisible = withinRange and minX ~= nil
                if objs.BoxFrame then
                    objs.BoxFrame.Visible = State.Features.BoxESP and boundsVisible
                    if boundsVisible and State.Features.BoxESP then
                        objs.BoxFrame.Position = UDim2.fromOffset(minX, minY)
                        objs.BoxFrame.Size = UDim2.fromOffset(math.max(1, maxX - minX), math.max(1, maxY - minY))
                        objs.BoxStroke.Color = CONFIG.Colors.BoxESP
                    end
                end
                if objs.HealthBar then
                    objs.HealthBar.Visible = State.Features.Healthbar and boundsVisible
                    if boundsVisible and State.Features.Healthbar then
                        local h = char:FindFirstChildOfClass("Humanoid")
                        local hp = h and math.max(0, h.Health) or 0
                        local maxHp = h and math.max(1, h.MaxHealth) or 1
                        local fraction = math.clamp(hp / maxHp, 0, 1)
                        local barHeight = math.max(12, maxY - minY)
                        objs.HealthBar.Position = UDim2.fromOffset(math.max(0, minX - 7), minY)
                        objs.HealthBar.Size = UDim2.fromOffset(4, barHeight)
                        objs.HealthFill.Size = UDim2.new(1, 0, fraction, 0)
                        objs.HealthFill.BackgroundColor3 = fraction > 0.5 and CONFIG.Colors.HealthGood or CONFIG.Colors.HealthLow
                    end
                end

                if State.Features.ESP then
                    objs.Highlight.Enabled = withinRange or State.Features.Chams
                    objs.Billboard.Enabled = withinRange
                    objs.Billboard.MaxDistance = State.Sliders.ESPDistance
                    objs.DistLabel.Text = string.format("%d studs", dist)
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if objs.HealthLabel and hum then
                        local hp = math.max(0, hum.Health)
                        local maxHp = math.max(1, hum.MaxHealth)
                        objs.HealthLabel.Text = string.format("HP %d / %d", math.floor(hp + 0.5), math.floor(maxHp + 0.5))
                    end
                else
                    objs.Billboard.Enabled = false
                end

                if State.Features.SkeletonESP and ((#(objs.SkeletonSegments or {}) == 0) or not objs.SkeletonHead or not objs.SkeletonHead.Parent) and char.Parent then
                    RebuildSkeleton(plr)
                end

                if State.Features.SkeletonESP and withinRange and objs.SkeletonHead then
                    local headPart = char:FindFirstChild("Head")
                    if headPart and headPart:IsA("BasePart") then
                        local sh, headOnScreen = Camera:WorldToViewportPoint(headPart.Position + Vector3.new(0, 0.15, 0))
                        if headOnScreen then
                            objs.SkeletonHead.Position = UDim2.fromOffset(sh.X, sh.Y)
                            objs.SkeletonHead.BackgroundColor3 = CONFIG.Colors.Skeleton
                            objs.SkeletonHead.Visible = true
                        else
                            objs.SkeletonHead.Visible = false
                        end
                    else
                        objs.SkeletonHead.Visible = false
                    end
                elseif objs.SkeletonHead then
                    objs.SkeletonHead.Visible = false
                end

                for _, segment in ipairs(objs.SkeletonSegments or {}) do
                    local frame = segment.Frame
                    local a = char:FindFirstChild(segment.A)
                    local b = char:FindFirstChild(segment.B)
                    if frame and a and b and a:IsA("BasePart") and b:IsA("BasePart") and State.Features.SkeletonESP and withinRange then
                        local sa, ona = Camera:WorldToViewportPoint(a.Position)
                        local sb, onb = Camera:WorldToViewportPoint(b.Position)
                        if ona and onb then
                            local dx = sb.X - sa.X
                            local dy = sb.Y - sa.Y
                            local length = math.sqrt(dx*dx + dy*dy)
                            if length > 1 then
                                frame.BackgroundColor3 = CONFIG.Colors.Skeleton
                                frame.Size = UDim2.fromOffset(2, length)
                                frame.Position = UDim2.fromOffset((sa.X + sb.X)/2, (sa.Y + sb.Y)/2)
                                frame.Rotation = math.deg(math.atan2(-dx, dy))
                                frame.Visible = true
                            else
                                frame.Visible = false
                            end
                        else
                            frame.Visible = false
                        end
                    elseif frame then
                        frame.Visible = false
                    end
                end

                if State.Features.Tracers and withinRange then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        local originX, originY = viewportSize.X / 2, viewportSize.Y
                        local dx, dy = screenPos.X - originX, screenPos.Y - originY
                        local length = math.sqrt(dx * dx + dy * dy)
                        local angle = math.deg(math.atan2(dy, dx))
                        objs.Tracer.Visible = true
                        objs.Tracer.Size = UDim2.fromOffset(length, 2)
                        objs.Tracer.Position = UDim2.fromOffset((originX + screenPos.X) / 2, (originY + screenPos.Y) / 2)
                        objs.Tracer.Rotation = angle
                    else
                        objs.Tracer.Visible = false
                    end
                else
                    objs.Tracer.Visible = false
                end
            end
        end
    end
end)

Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
    Camera = Workspace.CurrentCamera
    if Camera then
        Camera.FieldOfView = State.Sliders.CameraFOV
    end
end)
